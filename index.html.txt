function toMorse(text) {

    return text
        .toUpperCase()
        .split("")
        .map(function(character) {

            if (
                character === " "
            ) {

                return "/";

            }

            if (
                character === "," ||
                character === "."
            ) {

                return "";

            }

            return morseTable[character] || "";

        })
        .filter(function(value) {

            return value !== "";

        })
        .join(" ");

}

function getRemainingPetals() {

    return petals.filter(
        function(petal) {

            return !fallenPetals.has(
                petal.dataset.petal
            );

        }
    );

}


function losePetal(
    petal,
    showFinal = true
) {

    if (!petal) {
        return;
    }


    const id =
        petal.dataset.petal;


    if (
        fallenPetals.has(id)
    ) {

        return;

    }


    fallenPetals.add(id);


    petal.classList.remove(
        "returning"
    );


    void petal.offsetWidth;


    petal.classList.add(
        "falling"
    );


    if (
        getRemainingPetals().length === 0 &&
        showFinal
    ) {

        allPetalsGone();

    }

}


function loseOnePetal(
    showFinal = true
) {

    const remaining =
        getRemainingPetals();


    if (
        remaining.length === 0
    ) {

        return;

    }


    const petal =
        remaining[
            Math.floor(
                Math.random() *
                remaining.length
            )
        ];


    losePetal(
        petal,
        showFinal
    );

}

            );

    );


    if (

        !finalMessageShown
    ) {

        finalMessageShown =
            true;


        setTimeout(

                finalMessage.classList.add(
                    "visible"
                );

    }

}


/* =========================================================
   TREVAS
========================================================= */

function activateDarkness() {

    specialResponseUsed =
        true;


    const remaining =
        getRemainingPetals();


    if (
        remaining.length === 0
    ) {

        allPetalsGone();

        return;

    }


    remaining.forEach(
        function(petal, index) {

            setTimeout(
                function() {

                    losePetal(
                        petal,
                        false
                    );

                },
                index * 120
            );

        }
    );


    setTimeout(
        function() {

            container.classList.add(
                "darkness-fall"
            );

        },
        900
    );

}

/* =========================================================
   RESTAURAR PÉTALAS
========================================================= */

function restoreAllPetalsAnimated() {

    const fallen =
        petals.filter(
            function(petal) {

                return fallenPetals.has(
                    petal.dataset.petal
                );

            }
        );


    if (
        fallen.length === 0
    ) {

        return;

    }


    fallen.forEach(
        function(petal) {

            petal.classList.remove(
                "falling"
            );


            petal.style.opacity =
                "0";


            petal.style.transform =
                `
                translate(
                    ${getComputedStyle(petal)
                        .getPropertyValue("--fall-x")},
                    470px
                )
                rotate(
                    ${getComputedStyle(petal)
                        .getPropertyValue("--fall-rotation")}
                )
                `;


            void petal.offsetWidth;


            petal.classList.add(
                "returning"
            );


            setTimeout(
                function() {

                    petal.style.opacity =
                        "";

                    petal.style.transform =
                        "";

                },
                2500
            );

        }
    );


    setTimeout(
        function() {

            fallenPetals.clear();


            petals.forEach(
                function(petal) {

                    petal.classList.remove(
                        "falling",
                        "returning"
                    );

                    petal.style.opacity =
                        "";

                    petal.style.transform =
                        "";

                }
            );


            finalMessageShown =
                false;


            container.classList.remove(
                "darkness-fall"
            );


            finalMessage.classList.remove(
                "visible"
            );

        },
        2700
    );

}


/* =========================================================

========================================================= */

function resetClockAnimation() {

    clock.classList.remove(
        "forward",
        "backward"
    );


    void clock.offsetWidth;

}


function breakClock() {

    timeBroken =
        true;


    if (
        clockTimeout !== null
    ) {

        clearTimeout(
            clockTimeout
        );

        clockTimeout =
            null;

    }


    resetClockAnimation();


    clock.classList.add(
        "visible",
        "broken"
    );

}


function showClock() {

    if (
        timeBroken
    ) {

        return;

    }


    if (
        timeCount >= 4
    ) {

        breakClock();

        return;

    }


    resetClockAnimation();


    if (
        fallenPetals.size > 0
    ) {

        clock.classList.add(
            "visible",
            "backward"
        );


        setTimeout(
            function() {

                restoreAllPetalsAnimated();


                clock.classList.remove(
                    "visible",
                    "backward"
                );

            },
            1800
        );


        return;

    }


    clock.classList.add(
        "visible",
        "forward"
    );


    clockTimeout =
        setTimeout(
            function() {

                clock.classList.remove(
                    "visible",
                    "forward"
                );


                secondFlower.classList.add(
                    "visible"
                );


                setTimeout(
                    function() {

                        secondFlower.classList.add(
                            "falling"
                        );


                        setTimeout(
                            function() {

                                secondFlower.classList.add(
                                    "grounded"
                                );


                                setTimeout(
                                    function() {

                                        secondFlower.classList.add(
                                            "sprouting"
                                        );

                                    },
                                    300
                                );

                            },
                            1800
                        );

                    },
                    100
                );

            },
            5000
        );

}


/* =========================================================
   SEQUÊNCIA DOS QUATRO ELEMENTOS
========================================================= */

const elementSequence = [

    "luz",
    "energia",
    "conhecimento",
    "trevas"

];


function updateElementSequence(key) {

    if (
        key === elementSequence[0]
    ) {

        elementSequenceIndex =
            1;

        return false;

    }


    if (
        key === elementSequence[
            elementSequenceIndex
        ]
    ) {

        elementSequenceIndex++;

    } else {

        elementSequenceIndex =
            0;

    }


    if (
        elementSequenceIndex ===
        elementSequence.length
    ) {

        elementSequenceIndex =
            0;

        return true;

    }


    return false;

}

/* =========================================================
   EXIBIR RESPOSTA ESPECIAL
========================================================= */

function handleSpecialResponse(
    key
) {

    if (
        !Object.prototype.hasOwnProperty.call(
            specialResponses,
            key
        )
    ) {

        return false;

    }


    specialResponseUsed =
        true;


    /*
       Trevas possui comportamento próprio.
    */

    if (
        key === "trevas"
    ) {

        activateDarkness();

        return true;

    }


    showMessage(
        specialResponses[key]
    );


    return true;

}


/* =========================================================
   INPUT
========================================================= */

answer.addEventListener(
    "keydown",
    function(event) {

        if (
            event.key !== "Enter"
        ) {

            return;

        }


        if (
            treeEventActivated
        ) {

            return;

        }


        const value =
            answer.value.trim();


        const key =
            value.toLowerCase();


        inputCount++;


        clearInactivityTimers();


        if (
            message.textContent !== ""
        ) {

            clearMessageTimeout();

            message.textContent =
                "";

        }


        const isSpecial =
            Object.prototype.hasOwnProperty.call(
                specialResponses,
                key
            );


        const isName =
            nameGroup.includes(
                key
            );


        /*
           Especial já no primeiro input
           libera o botão.
        */

        if (
    inputCount >= 5 ||
    isSpecial
) {

    buttonUnlocked =
        true;

    button.classList.add(
        "available"
    );

}


        /* =================================================
           VITA
        ================================================= */

        if (
            key === "vita"
        ) {

            window.location.href =
                "https://SEU-SITE-AQUI.com";

            return;

        }


        /* =================================================
           SEQUÊNCIA DOS ELEMENTOS
        ================================================= */

        const specialElementSequence =
            updateElementSequence(key);


        if (
            specialElementSequence
        ) {

            specialResponseUsed =
                true;


            activateTreeEvent();

            return;

        }


        /* =================================================
           LUZ
        ================================================= */

        if (
            key === "luz"
        ) {

            specialResponseUsed =
                true;


            showMessage(
                specialResponses[key]
            );


            temporaryColor(
                "light"
            );


            return;

        }


        /* =================================================
           ENERGIA
        ================================================= */

        if (
            key === "energia"
        ) {

            specialResponseUsed =
                true;


            showMessage(
                specialResponses[key]
            );


            temporaryColor(
                "energy"
            );


            return;

        }


        /* =================================================
           CONHECIMENTO
        ================================================= */

        if (
            key === "conhecimento"
        ) {

            specialResponseUsed =
                true;


            showMessage(
                specialResponses[key]
            );


            temporaryColor(
                "knowledge"
            );


            loseOnePetal(
                true
            );


            return;

        }


        /* =================================================
           TREVAS NORMAL
        ================================================= */

        if (
            key === "trevas"
        ) {

            activateDarkness();

            return;

        }


        /* =================================================
           TEMPO
        ================================================= */

        if (
            key === "tempo"
        ) {

            timeCount++;


            if (
                timeCount >= 4
            ) {

                breakClock();

                return;

            }


            if (
                !timeBroken
            ) {

                showClock();

            }


            return;

        }


        /* =================================================
           YULLA KYRIARCHOS
        ================================================= */

        if (
            key === "yulla kyriarchos"
        ) {

            specialResponseUsed =
                true;


            showMessage(
                specialResponses[key]
            );


            temporaryColor(
                "knowledge"
            );


            return;

        }


        /* =================================================
           COMANDOS QUE PERDEM PÉTALA
        ================================================= */

        if (
            onePetalCommands.includes(
                key
            )
        ) {

            specialResponseUsed =
                true;


            showMessage(
                specialResponses[key]
            );


            loseOnePetal(
                true
            );


            return;

        }


        /* =================================================
           NOMES
        ================================================= */

        if (
    isName
) {

    if (
        namePromptShown
    ) {

        showMessage(
            "Não o de seu mestre. E sim o SEU nome."
        );

        namePromptShown =
            false;

        nameHintUsed =
            true;

        return;

    }

    showMessage(
        "..."
    );

    return;

}


        /* =================================================
           PRIMEIRO INPUT DESCONHECIDO
        ================================================= */

        if (
    inputCount === 1 &&
    !isSpecial &&
    !isName
) {

    namePromptShown =
        true;


    showMessage(
        "Comece pelo seu nome"
    );


    return;

}


        /* =================================================
           RESPOSTA ESPECIAL
        ================================================= */

        if (
            handleSpecialResponse(
                key
            )
        ) {

            return;

        }


        /* =================================================
           QUALQUER OUTRA RESPOSTA
        ================================================= */

        showMessage(
            "..."
        );

    }
);

        revealed ||
        treeEventActivated
    ) {

        return;

        !buttonUnlocked
    ) {

        return;

    }


    revealed =
        true;


    button.classList.add(
        "pressed"
    );


    revelation.classList.add(
        "visible"
    );

}

);

        ) {

        }

    }
);
