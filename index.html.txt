<script>

    /* =========================

    ========================= */

    const answer = document.getElementById("answer");
    const message = document.getElementById("message");


    answer.addEventListener("keydown", function(event) {

        if (event.key === "Enter") {

            const value = answer.value.trim();

            /*
             * A palavra correta.
             */

            if (value === "Vita") {

                /*
                 * TROQUE ESTE ENDEREÇO PELO SITE
                 * QUE VAMOS CRIAR DEPOIS.
                 */

                window.location.href = "https://SEU-SITE-AQUI.com";

            } else {

                message.textContent = "...";

                setTimeout(() => {
                    message.textContent = "";
                }, 2000);

            }

        }

    });


    /* =========================
       ALAVANCA
    ========================= */

    const leverContainer = document.getElementById("leverContainer");
    const lever = document.getElementById("lever");
    const morse = document.getElementById("morse");


    let pulled = false;


    leverContainer.addEventListener("click", function() {

        if (pulled) {
            return;
        }

        pulled = true;

        /*
         * Puxa a alavanca.
         */

        lever.classList.remove("normal");
        lever.classList.add("pulled");


        /*
         * Revela o código Morse.
         */

        setTimeout(() => {

            morse.classList.add("visible");

        }, 500);

    });

</script>
